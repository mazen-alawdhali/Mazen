    window.updateVisitorCount = async function() {
        const counterElement = document.getElementById('visitor-count');
        const t = dict[window.currentLang];
        if (!counterElement) return;

        if (!navigator.onLine) {
            counterElement.textContent = t.visitorCountOffline;
            counterElement.style.color = '#f59e0b'; 
            return;
        }

        try {
            const response = await fetch('https://api.counterapi.dev/v1/mazen_solar_calc_v7/visits/up');
            if (!response.ok) throw new Error('Network response was not ok');
            const data = await response.json();
            counterElement.textContent = (data.count + 9000).toLocaleString('en-US');
            counterElement.style.color = '#10b981'; 
        } catch (error) {
            counterElement.textContent = t.visitorCountError;
            counterElement.style.color = '#ef4444'; 
        }
    };

    function getCableSizeFromAmps(amps) {
        if (amps <= 50) return "25";
        if (amps <= 100) return "35";
        if (amps <= 150) return "50";
        if (amps <= 200) return "70";
        if (amps <= 250) return "95";
        if (amps <= 300) return "120";
        if (amps <= 400) return "150";
        return "Busbars / 2x120"; 
    }
    
    function getPVCableSizeFromAmps(amps) {
        if (amps <= 20) return "4";
        if (amps <= 30) return "6";
        if (amps <= 50) return "10";
        if (amps <= 75) return "16";
        if (amps <= 100) return "25";
        return "Combiner Box";
    }
    
    function getStandardBreakerSize(amps) {
        const standardBreakers = [16, 20, 25, 32, 40, 50, 63, 80, 100, 125, 160, 200, 250, 300, 400, 500, 630];
        for (let size of standardBreakers) {
            if (size >= amps) return size;
        }
        return Math.ceil(amps); 
    }

    window.calculateResults = function() {

                let itemNight = w * count * nH * (duty/100);

                nightLoadWh += itemNight;
